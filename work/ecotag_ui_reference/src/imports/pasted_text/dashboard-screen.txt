  Future<void> _loadDashboardData() async {
    setState(() {
      _isLoading = true;
      _errorMessage = null;
    });

    try {
      final healthRows = await _supabase
          .from('plant')
          .select('plant_id, health_status(status_name)');

      int healthy = 0;
      int dying = 0;
      int dead = 0;

      for (final row in healthRows) {
        final status = row['health_status']?['status_name'] as String?;
        switch (status) {
          case 'alive':
            healthy++;
            break;
          case 'dying':
            dying++;
            break;
          case 'dead':
            dead++;
            break;
        }
      }

      final recentRows = await _supabase
          .from('plant')
          .select('''
            plant_id,
            variety,
            date_planted,
            plant_classification ( common_name, plant_type ),
            campus ( campus_name ),
            growth_status ( growth_stage ),
            health_status ( status_name )
          ''')
          .order('created_at', ascending: false)
          .limit(5);

      if (!mounted) return;

      setState(() {
        _totalPlants = healthRows.length;
        _healthyCount = healthy;
        _dyingCount = dying;
        _deadCount = dead;
        _recentPlants = List<Map<String, dynamic>>.from(recentRows);
        _isLoading = false;
      });
    } on PostgrestException catch (e) {
      // Surfaces the real Postgres/PostgREST error (RLS denial, bad
      // relationship name, etc.) instead of a generic failure message.
      debugPrint('DASHBOARD LOAD ERROR (Postgrest): ${e.message}');
      if (!mounted) return;
      setState(() {
        _errorMessage = 'Could not load dashboard data: ${e.message}';
        _isLoading = false;
      });
    } catch (e) {
      debugPrint('DASHBOARD LOAD ERROR: $e');
      if (!mounted) return;

      // "ClientException: Failed to fetch" means the HTTP request never
      // got a response at all — this is a network-level failure, not a
      // database error, so RLS/query fixes won't help. Most common causes:
      //   1. The Supabase project is paused (free-tier projects auto-pause
      //      after a period of inactivity and must be manually resumed
      //      from the Supabase dashboard).
      //   2. No internet connection on this device/browser.
      //   3. The project URL in SupabaseConfig is wrong or the project
      //      was deleted/recreated (new URL, same-looking key).
      final isNetworkFailure = e.toString().contains('Failed to fetch') ||
          e.toString().contains('ClientException');

      setState(() {
        _errorMessage = isNetworkFailure
            ? 'Could not reach Supabase. Check that your project is not '
                'paused (Supabase dashboard \u2192 Project) and that this '
                'device has an internet connection.\n\nDetails: $e'
            : 'Could not load dashboard data: $e';
        _isLoading = false;
      });
    }
  }

    final parts = name.split(RegExp(r'[ ,]+')).where((s) => s.isNotEmpty);

  }

  DataRow _buildTableRow(
    String plant,
    String type,
    String variety,
    String campus,
    String stage,
    String health,
    String date,
  ) {
    final healthColor = health == 'Healthy'
        ? const Color(0xFF2E7D32)
        : health == 'At Risk'
            ? const Color(0xFFF57C00)
            : const Color(0xFFD32F2F);

    return DataRow(
      cells: [
        DataCell(Row(
          children: [
            Icon(Icons.local_florist, size: 16, color: Colors.green.shade300),
            const SizedBox(width: 8),
            Text(plant, style: const TextStyle(fontWeight: FontWeight.w600)),
          ],
        )),
